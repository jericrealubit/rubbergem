CREATE OR REPLACE FUNCTION reset_shift_log(p_shift_id TEXT)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  -- Clear the live_log rows and shift_messages chat history for this shift.
  -- Archiving to production_logs now happens live on each cycle submit
  -- (see components/PressForm.tsx), so this function only wipes the live log
  -- and the chat, both of which are scoped to the shift that's ending.
  --
  -- SECURITY DEFINER: live_log's RLS policies have no DELETE policy for the
  -- authenticated role, so a plain SECURITY INVOKER delete here was silently
  -- matching 0 rows (no error, no rows deleted). Running as the function
  -- owner bypasses RLS for this deliberate, gated destructive action.
  -- shift_messages likewise has no DELETE policy (see shift_messages.sql) --
  -- clearing it here reuses the same SECURITY DEFINER bypass.
  DELETE FROM public.live_log
  WHERE shift_id::text = p_shift_id;

  DELETE FROM public.shift_messages
  WHERE shift_id::text = p_shift_id;
END;
$$;

-- Lock execution down to logged-in operators only (mirrors the UI gating in
-- app/ProductionTable.tsx, which already disables the button unless a
-- session exists).
REVOKE ALL ON FUNCTION reset_shift_log(TEXT) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION reset_shift_log(TEXT) TO authenticated;
